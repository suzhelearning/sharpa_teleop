cmake_minimum_required(VERSION 3.24)
project(sharpa_manus_ros LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_library(USB_LIBRARY NAMES usb-1.0 REQUIRED)
find_library(UDEV_LIBRARY NAMES udev REQUIRED)
if(NOT EXISTS "${MANUS_SDK}/include/ManusSDK.h" OR NOT EXISTS "${MANUS_SDK}/lib/libManusSDK_Integrated.so")
  message(FATAL_ERROR "Set MANUS_SDK to a compatible client/ManusSDK directory")
endif()
add_executable(manus_ros manus_ros.cpp)
target_include_directories(manus_ros PRIVATE "${MANUS_SDK}/include")
ament_target_dependencies(manus_ros rclcpp geometry_msgs)
target_link_libraries(manus_ros Eigen3::Eigen Threads::Threads
  "${MANUS_SDK}/lib/libManusSDK_Integrated.so" ${USB_LIBRARY} ${UDEV_LIBRARY})
set_target_properties(manus_ros PROPERTIES BUILD_RPATH "${MANUS_SDK}/lib")

include(CTest)
if(BUILD_TESTING)
  add_executable(manus_pose_test test_manus_pose.cpp)
  target_include_directories(manus_pose_test PRIVATE "${MANUS_SDK}/include")
  ament_target_dependencies(manus_pose_test geometry_msgs)
  target_link_libraries(manus_pose_test Eigen3::Eigen)
  add_test(NAME manus_pose_transform COMMAND manus_pose_test)
endif()
